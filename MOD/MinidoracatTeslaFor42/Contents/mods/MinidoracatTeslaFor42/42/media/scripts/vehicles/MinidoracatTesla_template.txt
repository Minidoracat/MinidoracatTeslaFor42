module Base
{
    /* Shared EV base for every Tesla. Numbers, sounds and the part list; geometry lives in each car's own script. */
    /* Based on template vehicle ModernCar (generated/vehicles/vehicle_car_modern_template.txt) without Muffler. */
    template vehicle MinidoracatTeslaBase
    {
        mechanicType = 3,
        engineRepairLevel = 6,
        playerDamageProtection = 1.1,
        engineRPMType = firebird,
        /* mechanics window car diagram (VehicleScript.java:278-279, ISVehicleMechanics.lua:914-918); it only draws parts the car has */
        carMechanicsOverlay = Base.ModernCar,
        /* Tesla recordings (scripts/sounds/MinidoracatTesla_sounds.txt); the driver's motor whine and forward PWS are Lua (client/MinidoracatTesla/Sound.lua) */
        sound
        {
            alarmLoop = VehicleAlarmA VehicleAlarmB VehicleAlarmC VehicleAlarmD VehicleAlarmE,
            backSignal = MinidoracatTesla_PWSReverse,
            engine = MinidoracatTesla_Hum,
            engineStart = MinidoracatTesla_Contactors,
            engineTurnOff = MinidoracatTesla_ParkBrake,
            horn = VehicleHornStandard,
            ignitionFail = MinidoracatTesla_KeyAlert,
            ignitionFailNoPower = MinidoracatTesla_NoPower,
        }
        mass = 800,
        /* default only: MinidoracatTesla.Create.Motor sets SR 3000 / LR 5000 by pack (Core.lua PackPower) */
        engineForce = 5000,
        maxSpeed = 90.0,
        /* effective = script / 2 (BaseVehicle.java:9120), x2 with no Muffler (:9174-9175) = 10: every update stays at the 8-tile floor,
           random peaks 12 (cruise) to 24 (full throttle) tiles; ModernCar 80 with a new muffler = 32-40 */
        engineLoudness = 10,
        engineQuality = 100,
        stoppingMovementForce = 4.0,
        rollInfluence = 1.0,
        steeringIncrement = 0.04,
        steeringClamp = 0.3,
        suspensionStiffness = 40.0,
        suspensionCompression = 3.83,
        suspensionDamping = 2.88,
        maxSuspensionTravelCm = 10.0,
        suspensionRestLength = 0.2,
        wheelFriction = 1.4,
        frontEndHealth = 150,
        rearEndHealth = 150,
        template = TrunkDoor,
        template = Trunk/part/TruckBed,
        template = Seat/part/SeatFrontLeft,
        template = Seat/part/SeatFrontRight,
        template = Seat/part/SeatRearLeft,
        template = Seat/part/SeatRearRight,
        template = GloveBox,
        template = GasTank,
        /* 12V battery: tryStartEngine dereferences it without a null check (BaseVehicle.java:7570-7612) */
        template = Battery,
        template = Engine,
        template = EngineDoor,
        template = Heater,
        template = Windshield/part/Windshield,
        template = Windshield/part/WindshieldRear,
        template = Window/part/WindowFrontLeft,
        template = Window/part/WindowFrontRight,
        template = Window/part/WindowRearLeft,
        template = Window/part/WindowRearRight,
        template = Door/part/DoorFrontLeft,
        template = Door/part/DoorFrontRight,
        template = Door/part/DoorRearLeft,
        template = Door/part/DoorRearRight,
        template = Tire,
        template = Brake,
        template = Suspension,
        template = Radio,
        template = Headlight,
        /* drive battery: GasTank part with its own content type, so vanilla refuel menus skip it (ISVehicleMenu.lua:1062) */
        part GasTank
        {
            category = engine,
            specificItem = false,
            durability = 5,
            container
            {
                test = Vehicles.ContainerAccess.GasTank,
                contentType = Electricity,
                conditionAffectsCapacity = false,
            }
            lua
            {
                create = MinidoracatTesla.Create.Battery,
                update = MinidoracatTesla.Update.Battery,
                checkEngine = MinidoracatTesla.CheckEngine.Battery,
            }
            table install
            {
                skills = Mechanics:7,
                recipes = Advanced Mechanics,
                complete = MinidoracatTesla.InstallComplete.Battery,
            }
            /* an empty value deletes the key from the merged table (VehicleScript.java:1141-1145): no requireEmpty from */
            /* template_gastank.txt, so the pack comes out charged and the charge rides on the item */
            table uninstall
            {
                skills = Mechanics:7,
                recipes = Advanced Mechanics,
                requireEmpty = ,
            }
        }
        /* cargo by the 2025+ owner's manual (Dimensions): trunk 835 L behind the 2nd row, frunk 114 L. Anchor: Model Y */
        /* is a midsize SUV like vanilla SUV, whose trunk is NormalTrunk2 = 85 (mechanicType 3 alone would give the */
        /* ModernCar sedan's NormalTrunk3 = 45). A script capacity becomes the new trunk item's MaxCapacity */
        /* (VehicleUtils.createPartInventoryItem, Vehicles.lua:1181-1183; saved per item, InventoryItem.java:1803,2129). */
        part TruckBed
        {
            container
            {
                capacity = 85,
            }
        }
        /* frunk 114 / 835 L x 85 = 11.6 -> 12. A container with no contentType gets an ItemContainer from the script */
        /* capacity, item or not (VehicleParts.java:307-313); no item, so condition does not shrink it */
        part Frunk
        {
            area = Engine,
            category = bodywork,
            container
            {
                capacity = 12,
                test = MinidoracatTesla.ContainerAccess.Frunk,
                soundMap = ContainerPut VehicleTrunkTransferItem,
                soundMap = ContainerTake VehicleTrunkTransferItem,
            }
        }
        part Engine
        {
            lua
            {
                create = MinidoracatTesla.Create.Motor,
                update = MinidoracatTesla.Update.Engine,
                checkEngine = Vehicles.CheckEngine.Engine,
            }
        }
        part TrunkDoor
        {
            table install
            {
                recipes = Advanced Mechanics,
            }
            /* the rear glass is part of the tailgate's mesh: glass off first, as vanilla doors with their windows */
            table uninstall
            {
                recipes = Advanced Mechanics,
                requireUninstalled = WindshieldRear,
            }
        }
        part Seat*
        {
            table install
            {
                recipes = Advanced Mechanics,
            }
            table uninstall
            {
                recipes = Advanced Mechanics,
            }
        }
        part EngineDoor
        {
            table install
            {
                recipes = Advanced Mechanics,
            }
            table uninstall
            {
                recipes = Advanced Mechanics,
            }
        }
        part Heater
        {
            category = engine,
            lua
            {
                update = MinidoracatTesla.Update.Heater,
            }
        }
        /* 12V: vanilla update plus the DC-DC top-up from the drive pack while it holds >= 5% (Battery.lua B.dcdc) */
        part Battery
        {
            lua
            {
                create = MinidoracatTesla.Create.Battery12V,
                update = MinidoracatTesla.Update.Battery12V,
            }
        }
        part PassengerCompartment
        {
            category = nodisplay,
            lua
            {
                update = Vehicles.Update.PassengerCompartment,
            }
        }
        /* camp cabin light (CampPower.lua): the engine draws light parts like headlights. The id has no "Headlight", */
        /* so the headlight switch leaves it alone (VehicleParts.java:373), and no "Rear" (IsoGameCharacter.java:17311) */
        part CabinLight
        {
            category = nodisplay,
            lua
            {
                init = MinidoracatTesla.Init.CabinLight,
            }
        }
        part Windshield*
        {
            table install
            {
                recipes = Advanced Mechanics,
            }
            table uninstall
            {
                recipes = Advanced Mechanics,
            }
        }
        part WindshieldRear
        {
            table install
            {
                requireInstalled = TrunkDoor,
            }
        }
        part Window*
        {
            anim Open
            {
                sound = VehicleWindowElectricOpen,
            }
            anim Close
            {
                sound = VehicleWindowElectricClose,
            }
            table install
            {
                recipes = Advanced Mechanics,
            }
            table uninstall
            {
                recipes = Advanced Mechanics,
            }
        }
        part Door*
        {
            table install
            {
                recipes = Advanced Mechanics,
            }
            table uninstall
            {
                recipes = Advanced Mechanics,
            }
        }
        /* brakes and suspension: vanilla skills (Mechanics 3, Basic Mechanics). A lua block replaces the whole set */
        /* (VehicleScript.java:965-966), so create is repeated; update halves the wear for regen (Battery.lua) */
        part Brake*
        {
            lua
            {
                create = Vehicles.Create.Brake,
                update = MinidoracatTesla.Update.Brakes,
            }
        }
    }
}
